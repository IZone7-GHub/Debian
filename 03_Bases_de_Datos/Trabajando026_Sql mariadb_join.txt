-- 1. Creamos la tabla NOMBRES
CREATE TABLE NOMBRES (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    NOMBRE VARCHAR(50) NOT NULL,
    APATERNO VARCHAR(50) NOT NULL,
    AMATERNO VARCHAR(50) NOT NULL,
    EDAD INT NOT NULL,
    CLAVEUNICA VARCHAR(15) GENERATED ALWAYS AS (
        CONCAT(
            LEFT(NOMBRE, 1),
            LEFT(APATERNO, 1),
            LEFT(AMATERNO, 1),
            EDAD,
            SUBSTRING(TELEFONO, -2)
        )
    ) STORED UNIQUE NOT NULL
);

1.- Consulta (Muestra todo NOMBRES y lo que encuentre de CONTACTOS)
SELECT 
    NOMBRES.ID, 
    NOMBRES.NOMBRE, 
    NOMBRES.APATERNO, 
    NOMBRES.AMATERNO, 
    NOMBRES.EDAD, 
    CONTACTOS.TELEFONO, 
    CONTACTOS.EMAIL, 
    NOMBRES.CLAVEUNICA
FROM NOMBRES
LEFT JOIN CONTACTOS 
    ON NOMBRES.CLAVEUNICA = CONTACTOS.CLAVEUNICA;

2.- Consulta (Únicamente los que hacen match completo)
SELECT 
    NOMBRES.ID, 
    NOMBRES.NOMBRE, 
    NOMBRES.APATERNO, 
    NOMBRES.AMATERNO, 
    NOMBRES.EDAD, 
    CONTACTOS.TELEFONO, 
    CONTACTOS.EMAIL, 
    NOMBRES.CLAVEUNICA
FROM NOMBRES
INNER JOIN CONTACTOS 
    ON NOMBRES.CLAVEUNICA = CONTACTOS.CLAVEUNICA;

3.- Consulta (Actualizar TELEFONO en la tabla CONTACTOS tomando datos de NOMBRES)
Validamos los registros de la tabla CONTACTOS
SELECT * FROM CONTACTOS;

UPDATE CONTACTOS
INNER JOIN NOMBRES ON CONTACTOS.CLAVEUNICA = NOMBRES.CLAVEUNICA
SET CONTACTOS.EDAD = NOMBRES.EDAD
WHERE CONTACTOS.EDAD IS NULL OR CONTACTOS.EDAD = 0;

SELECT * FROM CONTACTOS;

4.- Consulta (Actualizar APATERNO y AMATERNO en la tabla CONTACTOS tomando datos de NOMBRES)
SELECT * FROM CONTACTOS;
UPDATE CONTACTOS
INNER JOIN NOMBRES ON CONTACTOS.CLAVEUNICA = NOMBRES.CLAVEUNICA
SET CONTACTOS.APATERNO = NOMBRES.APATERNO,
    CONTACTOS.AMATERNO = NOMBRES.AMATERNO
WHERE CONTACTOS.APATERNO IS NULL OR CONTACTOS.APATERNO = '';
SELECT * FROM CONTACTOS;

5.- Consulta (Actualizar TELEFONO y EMAIL en la tabla NOMBRES tomando datos de CONTACTOS)
SELECT * FROM NOMBRES;
UPDATE NOMBRES
INNER JOIN CONTACTOS ON NOMBRES.CLAVEUNICA = CONTACTOS.CLAVEUNICA
SET NOMBRES.TELEFONO = CONTACTOS.TELEFONO,
    NOMBRES.EMAIL = CONTACTOS.EMAIL
WHERE NOMBRES.TELEFONO IS NULL OR NOMBRES.TELEFONO = '';
SELECT * FROM NOMBRES;

Consultas para reforzar el tema:
-- 1.- Filtrar mayores o iguales a 30 años
SELECT * FROM NOMBRES WHERE EDAD >= 30;

-- 2.- Filtrar menores o iguales a 30 años
SELECT * FROM NOMBRES WHERE EDAD <= 30;

-- 3.- Buscar por nombre exacto ANA
SELECT * FROM NOMBRES WHERE NOMBRE = 'ANA';

-- 4.- Buscar nombres que comiencen con la letra A
SELECT * FROM NOMBRES WHERE NOMBRE LIKE 'A%';

-- 5.- Buscar nombres que contengan la letra A
SELECT * FROM NOMBRES WHERE NOMBRE LIKE '%A%';

-- 6.- Ordenar por edad de menor a mayor
SELECT * FROM NOMBRES ORDER BY EDAD ASC;

-- 7.- Ordenar por edad de mayor a menor
SELECT * FROM NOMBRES ORDER BY EDAD DESC;

-- 8.- Mostrar solo los primeros 2 registros
SELECT * FROM NOMBRES LIMIT 2;

-- 9.- Agregar el campo CIUDAD a la tabla NOMBRES
SELECT * FROM NOMBRES;
ALTER TABLE NOMBRES ADD CIUDAD VARCHAR(50) NOT NULL DEFAULT '' AFTER EMAIL;
SELECT * FROM NOMBRES;

-- 10.- Actualizar la ciudad a CDMX para la CLAVEUNICA A58
UPDATE NOMBRES SET CIUDAD = 'CDMX' WHERE CLAVEUNICA = 'A58';
SELECT * FROM NOMBRES;

-- 11.- Actualizar a TODOS los registros la ciudad a CDMX
UPDATE NOMBRES SET CIUDAD = 'MX';
SELECT * FROM NOMBRES;

-- 12.- Actualizar la edad de ANA a 50 años
UPDATE NOMBRES SET EDAD = 50 WHERE NOMBRE = 'ANA';
SELECT * FROM NOMBRES;

-- 13.- Actualizar el nombre de ANA a ANA REBECA
UPDATE NOMBRES SET NOMBRE = 'ANA REBECA' WHERE NOMBRE = 'ANA';
SELECT * FROM NOMBRES;

-- 14.- Contar cuántas personas tienen exactamente 50 años
SELECT COUNT(*) FROM NOMBRES WHERE EDAD = 50;
SELECT * FROM NOMBRES;

-- 15.- Filtrar edades que se repiten más de una vez
SELECT EDAD, COUNT(*) FROM NOMBRES GROUP BY EDAD HAVING COUNT(*) > 1;
SELECT * FROM NOMBRES;

-- 16.- Eliminar el registro de ANA REBECA
DELETE FROM NOMBRES WHERE NOMBRE = 'ANA REBECA';
SELECT * FROM NOMBRES;

-- 17.- Eliminar la tabla CONTACTOS Y NOMBRES
DROP TABLE CONTACTOS, NOMBRES;
